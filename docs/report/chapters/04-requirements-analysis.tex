\section{السرد النصي لحالات الاستخدام}

\subsection{حالات الاستخدام للمستخدمين غير المسجلين}

\subsubsection{السرد النصي لحالة الاستخدام طلب تسجيل مستخدم جديد}

\begin{enumerate}
  \item الاسم: طلب تسجيل مستخدم جديد
  \item الفاعل: مستخدم مجهول (غير مسجل)
  \item الوصف: يسمح للمستخدم المجهول بإرسال بياناته الشخصية وكلمة المرور للنظام لطلب حساب جديد (طالب أو معلم).
  \item الشروط المسبقة
  \begin{enumerate}
    \item أن يكون المستخدم متواجد في صفحة التسجيل الخاصة بالمنصة.
    \item ألا يكون لدى المستخدم جلسة دخول نشطة.
  \end{enumerate}
  \item الشروط اللاحقة
  \begin{enumerate}
    \item في حالة النجاح: يتم إنشاء سجل مستخدم جديد في قاعدة البيانات بحالة "معلق" (\en{Pending})، ويتم إرسال تنبيه عبر نظام الرسائل.
    \item في حالة الفشل: لا يتم إنشاء أي مستخدم؛ ويعرض النظام رسالة خطأ مناسبة.
  \end{enumerate}
  \item السيناريو الرئيسي (المسار الناجح)
\end{enumerate}

    \item 4: استعادة حساب محذوف: إذا كان البريد ينتمي لحساب محذوف سابقا، يقوم النظام بتحديث بيانات الحساب القديم وإعادته لحالة "معلق" بدلا من إنشاء سجل جديد.
    \item 6: اختيار معرف دور غير صحيح: إذا أدخل المستخدم معرف دور غير موجود، يعرض النظام خطأ في البيانات.
  \end{enumerate}
\end{enumerate}

\subsubsection{السرد النصي لحالة الاستخدام طلب إعادة تعيين كلمة المرور}

\begin{enumerate}
  \item الاسم: طلب إعادة تعيين كلمة المرور
  \item الفاعل: مستخدم مجهول (غير مسجل)
  \item الوصف: يسمح للمستخدم الذي نسي كلمة مروره بطلب رمز تحقق مؤقت مكون من 6 أرقام يتم إرساله إلى بريده الإلكتروني المسجل.
  \item الشروط المسبقة
  \begin{enumerate}
    \item أن يكون للمستخدم حساب مسبق في النظام.
    \item أن يتواجد المستخدم في صفحة "نسيت كلمة المرور".
  \end{enumerate}
  \item الشروط اللاحقة
  \begin{enumerate}
    \item في حالة النجاح: يتم تخزين رمز مشفر في قاعدة البيانات، وزيادة عداد المحاولات اليومية، وجدولة إرسال البريد الإلكتروني.
    \item في حالة الفشل: لا يتم إرسال بريد، ولكن النظام يظهر نفس رسالة النجاح (لأسباب أمنية) لمنع المخترقين من معرفة الحسابات الموجودة.
  \end{enumerate}
  \item السيناريو الرئيسي (المسار الناجح)
\end{enumerate}

    \item 5: تجاوز الحد اليومي: إذا حاول المستخدم طلب الرمز أكثر من 5 مرات في يوم واحد، يرفض النظام الطلب ويظهر رسالة: "تم الوصول للحد اليومي، يرجى المحاولة بعد 24 ساعة".
    \item 7: الرمز لا يزال صالحا: إذا طلب المستخدم رمزا جديدا وكان الرمز السابق لم تنته صلاحيته بعد، يقوم النظام بإصدار رمز جديد وإلغاء القديم فورا.
  \end{enumerate}
\end{enumerate}

\subsubsection{السرد النصي لحالة الاستخدام إعادة تعيين كلمة المرور}

\begin{enumerate}
  \item الاسم: إعادة تعيين كلمة المرور
  \item الفاعل: مستخدم مجهول (غير مسجل)
  \item الوصف: يسمح للمستخدم المجهول بإكمال عملية إعادة تعيين كلمة مرور الحساب عن طريق إدخال رمز التحقق المكون من 6 أرقام وكلمة المرور الجديدة.
  \item الشروط المسبقة:
  \begin{enumerate}
    \item أن يكون المستخدم قد طلب رمز إعادة تعيين بنجاح مسبقا.
    \item أن يكون لدى المستخدم رمز إعادة تعيين المرسل إلى بريده الإلكتروني.
  \end{enumerate}
  \item الشروط اللاحقة
  \begin{enumerate}
    \item في حالة النجاح: يتم تحديث كلمة المرور المشفرة في النظام، ويتم حذف رمز إعادة التعيين من النظام، ويصبح بإمكان المستخدم تسجيل الدخول ببياناته الجديدة.
    \item في حالة الفشل: تظل كلمة المرور القديمة كما هي، ويتم إبلاغ المستخدم بأن الرمز غير صحيح أو انتهت صلاحيته.
  \end{enumerate}
  \item السيناريو الرئيسي (المسار الناجح)
\end{enumerate}

    \item 3: الرمز غير موجود: إذا لم يجد النظام رمز استعادة صالح (ربما تم استخدامه سابقا أو لم يتم طلبه أصلا)، يظهر النظام خطأ.
    \item 4: انتهاء صلاحية الرمز: إذا حاول المستخدم إدخال الرمز بعد مرور 15 دقيقة، يرفض النظام الطلب.
    \item 5: الرمز غير صحيح: إذا أدخل المستخدم أرقاما لا تطابق الرمز المشفر المحفوظ، يرفض النظام العملية.
  \end{enumerate}
\end{enumerate}

\subsubsection{السرد النصي لحالة الاستخدام تسجيل الدخول}

\begin{enumerate}
  \item الاسم: تسجيل الدخول
  \item الفاعل: مستخدم مجهول (غير مسجل)
  \item الوصف: يسمح للمستخدم المسجل بالمصادقة مع النظام باستخدام بيانات اعتماده للحصول على رموز الوصول اللازمة للتفاعل مع المنصة.
  \item الشروط المسبقة
  \begin{enumerate}
    \item أن يكون للمستخدم حساب مسجل مسبقا في النظام.
    \item أن يتواجد المستخدم في صفحة تسجيل الدخول وألا يكون مسجلا دخوله حاليا.
  \end{enumerate}
  \item الشروط اللاحقة
  \begin{enumerate}
    \item في حالة النجاح: يقوم النظام بإنشاء وإرجاع رمز وصول ورمز تحديث، ويتم حفظ رمز التحديث لدى النظام.
    \item في حالة الفشل: لا يتم إصدار أي رموز، ويعرض النظام رسالة خطأ توضح سبب الفشل (بيانات غير صحيحة أو حالة الحساب).
  \end{enumerate}
  \item السيناريو الرئيسي (المسار الناجح)
\end{enumerate}

    \item 3: بيانات غير صحيحة: إذا كانت كلمة المرور غير مطابقة، يظهر النظام رسالة: "بيانات الاعتماد غير صحيحة".
    \item 3: الحساب مرفوض: إذا كانت حالة الحساب \en{Rejected} يظهر النظام رسالة: "طلب التسجيل الخاص بك تم رفضه."
    \item 4: الحساب قيد الانتظار: إذا كانت حالة الحساب \en{Pending} يظهر النظام رسالة بأن الحساب بانتظار الموافقة. يرجى انتظار تفعيل الحساب من قبل المسؤول.
    \item 4: الحساب معطل: إذا كانت حالة الحساب \en{Deactivated} يظهر النظام رسالة: "تم تعطيل هذا الحساب".
  \end{enumerate}
\end{enumerate}

  \begin{enumerate}

  \end{enumerate}
  \item السيناريو الرئيسي (المسار الناجح)
