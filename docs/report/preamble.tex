% ORDERING RULE — READ BEFORE ADDING A PACKAGE

% polyglossia loads bidi, and bidi refuses most packages loaded after it:
%     ! Package bidi Error: Oops! you have loaded package X after bidi package.
% EVERY new \usepackage goes in the block ABOVE polyglossia. Only language and
% font configuration goes below it.
%
% SIZING NOTE
% -----------
% The template sets Latin 12pt against Arabic 15pt — a ratio of 1.25, applied
% via Scale=1.25 on the Arabic font. So every size below is expressed as the
% Arabic point size divided by 1.25, and comes out correct on the page.
% ===========================================================================

\usepackage{listings}
\usepackage{xcolor}
\usepackage{booktabs}
\usepackage{longtable}
\usepackage{graphicx}
\usepackage{titlesec}
\usepackage{caption}
\usepackage{setspace}

            headsep=0.5in,footskip=0.5in]{geometry}
\usepackage{hyperref}

\usepackage{polyglossia}
\setmainlanguage[numerals=maghrib]{arabic}   % maghrib = Western digits 0-9
\setotherlanguage{english}

% --- Fonts (STRUCTURE.md §2) ----------------------------------------------
\setmainfont{Times New Roman}
\newfontfamily\arabicfont[Script=Arabic,Scale=1.25]{Traditional Arabic}
\newfontfamily\englishfont{Times New Roman}
% DEVIATION FROM THE TEMPLATE, DELIBERATE. The template specifies Cambria for
% code ("رماز برنامج"), but only Cambria Italic/Bold are installed here — there
% is no Regular — and Cambria is proportional, which is poor for the volume of
% code this report carries. A monospace face is used instead. Raise with the
% supervisor; switching back is one line if they insist.
\newfontfamily\codefont[Scale=0.85]{DejaVu Sans Mono}

% --- Body text (STRUCTURE.md §2) ------------------------------------------
\setlength{\parindent}{0.5in}   % 720 twips
\setlength{\parskip}{6pt}       % 120 twips "space after"
\setstretch{1.15}

% Justified Arabic prose carrying long unbreakable Latin identifiers
% (\code{GeminiEmbeddingProvider}) cannot always be broken within the measure,
% and TeX then lets the line run into the margin. emergencystretch authorises an
% extra pass with looser spacing, used ONLY on lines that would otherwise be
% overfull — ordinary paragraphs are unaffected.
\setlength{\emergencystretch}{3em}

% ===========================================================================
% Arabic ordinals — the template writes "الفصل الأول", not "الفصل 1".
% ===========================================================================
\newcommand{\arabicordinal}[1]{%
  \ifcase#1\relax
    \or الأول\or الثاني\or الثالث\or الرابع\or الخامس%
    \or السادس\or السابع\or الثامن\or التاسع\or العاشر%
  \else #1\fi}

% ===========================================================================
% Numbering (STRUCTURE.md §5)
%
% THE CRITICAL BIT — and it is counter-intuitive, so do not "fix" it.
%
% The template requires the CHAPTER number furthest RIGHT: section 2 of
% chapter 1 must appear as "2.1". Verified by rendering the .dotx itself.
%
% Under bidi, the period between digits is a neutral, so "1.2" is split into
% two runs which are then ordered right-to-left — the pair comes out visually
% REVERSED. LaTeX's DEFAULT \thesection is \thechapter.\arabic{section}, i.e.
% "1.2", which therefore renders as "2.1" — exactly the template's form.
%
% So we deliberately keep the defaults. Redefining them to "section.chapter"
% double-reverses and yields "1.2" on the page, which is wrong. This is the
% manual, error-prone step the template warns Word users about; here it is
% simply correct by default.
% ===========================================================================

% The template allows 4 levels (chapter + 3) and lists 2 levels in the contents.
% report's defaults are secnumdepth=2 (subsubsections come out UNNUMBERED) and
% tocdepth=2 (contents too deep), so both must be set explicitly.
\setcounter{secnumdepth}{3}
\setcounter{tocdepth}{1}

% Figures, tables and equations run sequentially across the WHOLE report —
% the chapter number must never appear in them (STRUCTURE.md §5).
\counterwithout{figure}{chapter}
\counterwithout{table}{chapter}
\counterwithout{equation}{chapter}

% ===========================================================================
% Headings (STRUCTURE.md §2). Sizes are Arabic-pt / 1.25.
% ===========================================================================
% Chapter: number line at 22pt Arabic, title at 32pt Arabic, both centred.
\titleformat{\chapter}[display]
  {\centering\bfseries}
  {\normalfont\fontsize{17.6}{22}\selectfont الفصل~\arabicordinal{\value{chapter}}}
  {12pt}
  {\fontsize{25.6}{31}\selectfont}
\titlespacing*{\chapter}{0pt}{6pt}{18pt}

% DEVIATION FROM THE TEMPLATE, DELIBERATE. The template writes the section
% number with a leading dash ("-2.1", STRUCTURE.md §5); the dash is dropped here
% because it reads as a stray hyphen before the heading text. Restoring it means
% putting `-` back in front of \thesection below. Raise with the supervisor.
\titleformat{\section}{\bfseries\fontsize{17.6}{22}\selectfont}{\thesection}{0.5em}{}
\titlespacing*{\section}{0pt}{18pt}{6pt}

\titleformat{\subsection}{\bfseries\fontsize{13.6}{17}\selectfont}{\thesubsection}{0.5em}{}
\titlespacing*{\subsection}{0pt}{12pt}{6pt}

\titleformat{\subsubsection}{\bfseries\fontsize{12.8}{16}\selectfont}{\thesubsubsection}{0.5em}{}
\titlespacing*{\subsubsection}{0pt}{12pt}{6pt}

% An unnumbered centred title at 24pt Arabic — used for الخلاصة, المحتويات,
% قائمة الأشكال, الخاتمة, المراجع and the other front/back matter headings.
\newcommand{\centredtitle}[1]{%
  \begin{center}\bfseries\fontsize{19.2}{24}\selectfont #1\end{center}%
  \vspace{12pt}}

% The contents/figure/table lists print their OWN heading. Style those to match
% \centredtitle rather than adding a second one above them, which duplicates it.
\renewcommand{\contentsname}{المحتويات}
\renewcommand{\listfigurename}{قائمة الأشكال}
\renewcommand{\listtablename}{قائمة الجداول}
\newcommand{\listtitlefont}{\hfill\bfseries\fontsize{19.2}{24}\selectfont}
\renewcommand{\cfttoctitlefont}{\listtitlefont}
\renewcommand{\cftloftitlefont}{\listtitlefont}
\renewcommand{\cftlottitlefont}{\listtitlefont}
\renewcommand{\cftaftertoctitle}{\hfill}
\renewcommand{\cftafterloftitle}{\hfill}
\renewcommand{\cftafterlottitle}{\hfill}

% The short summary paragraph that opens every chapter (STRUCTURE.md §4).
% Style "ملخص الفصل": italic, indented ~0.75in both sides, 36pt after.
%
% Centred rather than justified — a deliberate deviation from the template,
% which specifies justified. Do NOT use \centering here: it would overwrite the
% \leftskip/\rightskip margins set below. The infinitely stretchable glue added
% to both skips is what centres each line inside the inset measure.
\newenvironment{chaptersummary}
  {\par\itshape\parindent=0pt \parfillskip=0pt
   \leftskip=0.75in plus 1fil \rightskip=0.75in plus 1fil
   \relax\ignorespaces}
  {\par\vspace{36pt}}

% The transition sentence that must close every chapter (STRUCTURE.md §4).
\newcommand{\transition}[1]{\par\vspace{24pt}\noindent #1}

% ===========================================================================
% Captions (STRUCTURE.md §6): "الشكل 1- text", 14pt Arabic bold, centred.
% Figure caption goes BELOW the figure; table caption goes ABOVE the table.
% ===========================================================================
\DeclareCaptionLabelSeparator{hiast}{- }
\captionsetup{labelsep=hiast,justification=centering,font={bf,small},
              labelfont=bf,skip=6pt}
\captionsetup[figure]{name=الشكل}
\captionsetup[table]{name=الجدول}

% ===========================================================================
% Code (STRUCTURE.md §2): style "رماز برنامج" — Cambria, forced left-to-right.
% ===========================================================================
\lstset{
  basicstyle=\codefont\small,
  frame=single,
  columns=fullflexible,
  keepspaces=true,
  breaklines=true,
  showstringspaces=false,
  numbers=left,
  numberstyle=\tiny,
  xleftmargin=2em,
}
% Every listing MUST be wrapped in the `english` environment so it is typeset
% left-to-right inside RTL prose:
%
%     \begin{english}
%     \begin{lstlisting}[language=Java]
%     ...
%     \end{lstlisting}
%     \end{english}
%
% Do NOT fold this into an lstnewenvironment — listings reads its body
% verbatim, and the wrapper leaves `english` open, which swallows
% \end{document} and aborts the build with "no legal \end found".

% ===========================================================================
% Links: the template mandates BLACK ONLY (STRUCTURE.md §7). Links stay
% clickable but are not coloured.
% ===========================================================================
\hypersetup{colorlinks=true,linkcolor=black,urlcolor=black,citecolor=black,
            bookmarksnumbered=true}

% --- Convenience macros ----------------------------------------------------
% \en{...} wraps EVERY Latin-script term inside Arabic prose. Without it the
% bidi algorithm can reorder adjacent punctuation.
\newcommand{\en}[1]{\textenglish{#1}}
% \code{...} for an inline identifier.
\newcommand{\code}[1]{\textenglish{\codefont #1}}
